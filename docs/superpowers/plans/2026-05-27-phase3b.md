```javascript
// --- 棚卸セッションID生成 ---
function generateStocktakeSessionId() {
  const today = formatDate().replace(/-/g, '');
  const sheet = getSheet('T_棚卸セッション');
  const data = sheet.getDataRange().getValues();
  let maxNum = 0;
  const prefix = ID_PREFIX.STOCKTAKE + today + '-';
  for (let i = 1; i < data.length; i++) {
    const id = String(data[i][0] || '');
    if (id.startsWith(prefix)) {
      const num = parseInt(id.substring(prefix.length), 10);
      if (num > maxNum) maxNum = num;
    }
  }
  const next = String(maxNum + 1).padStart(3, '0');
  return prefix + next;
}

function startStocktake(params) {

  try {

    const sessionId = generateStocktakeSessionId();
    const now = formatDateTime();

    sheet.appendRow([sessionId, now, auth.userId, '進行中', '', '', '']);

// --- 棚卸カウント記録 ---
function recordStocktakeCount(params) {
  const auth = requireGmailAuth(params.gmail, params.password);

  if (!params.sessionId) throw new Error('セッションIDが必要です');
  if (!params.品目ID) throw new Error('品目IDが必要です');
  if (params.実数 === undefined || params.実数 === '') throw new Error('実数が必要です');

  const 実数 = Number(params.実数);
  if (isNaN(実数) || 実数 < 0) throw new Error('実数は0以上の数値を入力してください');

  // セッション存在・進行中チェック
  const sessions = getSheetData('T_棚卸セッション');
  const session = sessions.find(s => s['セッションID'] === params.sessionId);
  if (!session) throw new Error('セッションが見つかりません');
  if (session['ステータス'] !== '進行中') throw new Error('このセッションは既に終了しています');

  // 品目情報取得
  const item = getItemById(params.品目ID);
  if (!item) throw new Error('品目ID ' + params.品目ID + ' が見つかりません');

  // 帳簿在庫取得
  const 帳簿在庫 = getCurrentStockByItemId(params.品目ID);
  const 差異 = 実数 - 帳簿在庫;
  const now = formatDateTime();

  const lock = LockService.getScriptLock();
  if (!lock.tryLock(10000)) throw new Error('他の処理が実行中です');

  try {
    const sheet = getSheet('T_棚卸明細');
    const data = sheet.getDataRange().getValues();
    const headers = data[0];
    const sessionIdCol = headers.indexOf('セッションID');
    const itemIdCol = headers.indexOf('品目ID');

    // 同セッション・同品目の既存行を探す（上書き用）
    let existingRow = -1;
    for (let i = 1; i < data.length; i++) {
      if (data[i][sessionIdCol] === params.sessionId && data[i][itemIdCol] === params.品目ID) {
        existingRow = i + 1; // 1-indexed
        break;
      }
    }

    // T_棚卸明細列順: セッションID / 品目ID / 帳簿在庫 / 実数 / 差異 / カウント日時 / 備考
    const row = [params.sessionId, params.品目ID, 帳簿在庫, 実数, 差異, now, ''];

    if (existingRow > 0) {
      // 上書き
      sheet.getRange(existingRow, 1, 1, row.length).setValues([row]);
    } else {
      // 新規追加
      sheet.appendRow(row);
    }

    return {
      品名: item['品名'],
      帳簿在庫: 帳簿在庫,
      実数: 実数,
      差異: 差異,
    };
  } finally {
    lock.releaseLock();
  }
}

// --- 棚卸サマリー取得 ---
function getStocktakeSummary(params) {
  requireGmailAuth(params.gmail, params.password);

  if (!params.sessionId) throw new Error('セッションIDが必要です');

  const details = getSheetData('T_棚卸明細')
    .filter(d => d['セッションID'] === params.sessionId);

  const counted = details.filter(d => d['備考'] !== '未カウント').length;
  const withDifference = details.filter(d => d['備考'] !== '未カウント' && Number(d['差異']) !== 0).length;

  // 帳簿在庫>0の全品目数
  const items = getItems();
  const itemsWithStock = items.filter(item => getCurrentStockByItemId(item['品目ID']) > 0);
  const countedItemIds = new Set(details.map(d => d['品目ID']));
  const uncounted = itemsWithStock.filter(item => !countedItemIds.has(item['品目ID'])).length;

  return { counted: counted, withDifference: withDifference, uncounted: uncounted };
}

// --- 棚卸終了 ---
function endStocktake(params) {
  const auth = requireGmailAuth(params.gmail, params.password);

  if (!params.sessionId) throw new Error('セッションIDが必要です');

  // セッション進行中チェック
  const sessionSheet = getSheet('T_棚卸セッション');
  const sessionData = sessionSheet.getDataRange().getValues();
  const headers = sessionData[0];
  const sessionIdCol = headers.indexOf('セッションID');
  const statusCol = headers.indexOf('ステータス');
  const endTimeCol = headers.indexOf('終了日時');

  let sessionRow = -1;
  for (let i = 1; i < sessionData.length; i++) {
    if (sessionData[i][sessionIdCol] === params.sessionId) {
      if (sessionData[i][statusCol] !== '進行中') throw new Error('このセッションは既に終了しています');
      sessionRow = i + 1;
      break;
    }
  }
  if (sessionRow < 0) throw new Error('セッションが見つかりません');

  const lock = LockService.getScriptLock();
  if (!lock.tryLock(10000)) throw new Error('他の処理が実行中です');

  try {
    const now = formatDateTime();

    // 未カウント品目を差異ありとして追加
    const details = getSheetData('T_棚卸明細')
      .filter(d => d['セッションID'] === params.sessionId);
    const countedItemIds = new Set(details.map(d => d['品目ID']));

    const items = getItems();
    const detailSheet = getSheet('T_棚卸明細');
    const uncountedItems = [];

    for (const item of items) {
      if (countedItemIds.has(item['品目ID'])) continue;
      const stock = getCurrentStockByItemId(item['品目ID']);
      if (stock <= 0) continue;

      // T_棚卸明細列順: セッションID / 品目ID / 帳簿在庫 / 実数 / 差異 / カウント日時 / 備考
      detailSheet.appendRow([params.sessionId, item['品目ID'], stock, '', '', now, '未カウント']);
      uncountedItems.push({ 品目ID: item['品目ID'], 品名: item['品名'], 帳簿在庫: stock });
    }

    // セッションステータスを「完了待ち」に更新
    sessionSheet.getRange(sessionRow, statusCol + 1).setValue('完了待ち');
    sessionSheet.getRange(sessionRow, endTimeCol + 1).setValue(now);

    return { success: true, uncountedItems: uncountedItems };
  } finally {
    lock.releaseLock();
  }
}

// --- 棚卸セッション一覧取得（PC管理画面用） ---
function getStocktakeSessions(params) {
  requireGmailAuth(params.gmail, params.password);

  const sessions = getSheetData('T_棚卸セッション');
  const details = getSheetData('T_棚卸明細');
  const users = getSheetData('M_担当者');

  return sessions.map(s => {
    const sessionDetails = details.filter(d => d['セッションID'] === s['セッションID']);
    const counted = sessionDetails.filter(d => d['備考'] !== '未カウント').length;
    const withDiff = sessionDetails.filter(d => d['備考'] !== '未カウント' && Number(d['差異']) !== 0).length;
    const uncounted = sessionDetails.filter(d => d['備考'] === '未カウント').length;
    const user = users.find(u => u['担当者ID'] === s['担当者ID']);
    const approver = s['承認者ID'] ? users.find(u => u['担当者ID'] === s['承認者ID']) : null;

    return {
      セッションID: s['セッションID'],
      開始日時: s['開始日時'],
      担当者名: user ? user['氏名'] : s['担当者ID'],
      ステータス: s['ステータス'],
      終了日時: s['終了日時'] || '',
      承認者名: approver ? approver['氏名'] : '',
      承認日時: s['承認日時'] || '',
      カウント数: counted,
      差異あり数: withDiff,
      未カウント数: uncounted,
    };
  }).reverse(); // 新しい順
}

// --- 棚卸承認（PC管理画面用） ---
function approveStocktake(params) {
  const auth = requireGmailAuth(params.gmail, params.password);

  if (!params.sessionId) throw new Error('セッションIDが必要です');

  // セッション「完了待ち」チェック
  const sessionSheet = getSheet('T_棚卸セッション');
  const sessionData = sessionSheet.getDataRange().getValues();
  const headers = sessionData[0];
  const sessionIdCol = headers.indexOf('セッションID');
  const statusCol = headers.indexOf('ステータス');
  const approverCol = headers.indexOf('承認者ID');
  const approveTimeCol = headers.indexOf('承認日時');

  let sessionRow = -1;
  for (let i = 1; i < sessionData.length; i++) {
    if (sessionData[i][sessionIdCol] === params.sessionId) {
      if (sessionData[i][statusCol] !== '完了待ち') throw new Error('承認できるのは「完了待ち」のセッションのみです');
      sessionRow = i + 1;
      break;
    }
  }
  if (sessionRow < 0) throw new Error('セッションが見つかりません');

  // 差異があるレコードを取得
  const details = getSheetData('T_棚卸明細')
    .filter(d => d['セッションID'] === params.sessionId)
    .filter(d => {
      // 未カウント or 差異あり → 調整対象
      if (d['備考'] === '未カウント') return false; // 未カウントは調整しない（実数不明のため）
      return Number(d['差異']) !== 0;
    });

  const lock = LockService.getScriptLock();
  if (!lock.tryLock(10000)) throw new Error('他の処理が実行中です');

  try {
    const now = formatDateTime();
    let adjustedCount = 0;

    // 差異ありレコードごとにT_入出庫に「棚卸調整」追加
    for (const d of details) {
      appendInventoryRecord({
        区分: '棚卸調整',
        品目ID: d['品目ID'],
        数量: Number(d['差異']),
        備考: '棚卸セッション ' + params.sessionId,
        ステータス: '確定',
      }, auth);
      adjustedCount++;
    }

    // セッションステータスを「確定」に更新
    sessionSheet.getRange(sessionRow, statusCol + 1).setValue('確定');
    sessionSheet.getRange(sessionRow, approverCol + 1).setValue(auth.userId);
    sessionSheet.getRange(sessionRow, approveTimeCol + 1).setValue(now);

    return { adjustedCount: adjustedCount };
  } finally {
    lock.releaseLock();
  }
}
```

```javascript
  async function autoRefreshLocations() {
    if (!isApiConfigured()) return;
    const ts = localStorage.getItem('CACHED_LOCATIONS_TIMESTAMP');
    if (ts) {
      const elapsed = Date.now() - Number(ts);
      if (elapsed < 24 * 60 * 60 * 1000) return; // 24時間未満なら何もしない
    }
    // バックグラウンドで取得（UIブロックなし）
    try {
      const url = getApiUrl() + '?action=getLocations';
      const res = await fetch(url, { method: 'GET', redirect: 'follow' });
      const json = await res.json();
      if (json.success && json.data) {
        localStorage.setItem('CACHED_LOCATIONS', JSON.stringify(json.data));
        localStorage.setItem('CACHED_LOCATIONS_TIMESTAMP', String(Date.now()));
      }
    } catch (err) {
      console.log('Auto-refresh locations skipped:', err.message);
    }
  }
```

```javascript
    const gmail = document.getElementById('input-gmail').value.trim();

      return;

    localStorage.setItem('USER_GMAIL', gmail);
    localStorage.setItem('USER_PASSWORD', password);

    // API認証で担当者情報を取得
    if (isApiConfigured()) {
      try {
        const url = getApiUrl() + '?action=authenticateByGmail&gmail=' + encodeURIComponent(gmail) + '&password=' + encodeURIComponent(password);
        const res = await fetch(url, { method: 'GET', redirect: 'follow' });
        const json = await res.json();
        if (json.success && json.data && json.data.name) {
          localStorage.setItem('USER_NAME', json.data.name);
          localStorage.setItem('USER_ROLE', json.data.role || '');
        }
      } catch (err) {
        console.log('担当者情報の取得をスキップ:', err.message);
      }
    }

    updateGmailStatus();
    alert('ログイン情報を保存しました');
  }
```

```javascript
    if (!stDetailSessionId) return;
    if (!confirm('この棚卸セッションを承認し、在庫を調整しますか？\nこの操作は元に戻せません。')) return;

    const btn = document.getElementById('btn-st-approve');
    btn.disabled = true;
    btn.textContent = '処理中...';

    try {
      const gmail = localStorage.getItem('USER_GMAIL') || '';
      const password = localStorage.getItem('USER_PASSWORD') || '';
      const url = getApiUrl() + '?action=approveStocktake&gmail=' + encodeURIComponent(gmail)
        + '&password=' + encodeURIComponent(password)
        + '&sessionId=' + encodeURIComponent(stDetailSessionId);
      const res = await fetch(url, { method: 'GET', redirect: 'follow' });
      const json = await res.json();
      if (!json.success) throw new Error(json.error || '承認に失敗');

      alert('承認完了: ' + json.data.adjustedCount + '件の在庫調整を実行しました。');
      closeStDetailModal();
      fetchData(); // テーブル再読み込み
    } catch (err) {
      alert('エラー: ' + err.message);
    } finally {
      btn.disabled = false;
      btn.textContent = '承認して在庫調整';
    }
  }
```
