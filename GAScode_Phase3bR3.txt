/**
 * 現在の有効スプシを開く
 */
function getCurrentSpreadsheet() {
  return SpreadsheetApp.openById(getCurrentSpreadsheetId());
}

/**
 * 指定シートを取得
 * @param {string} sheetName SHEETS定数を渡す
 */
function getSheet(sheetName) {
  const sheet = getCurrentSpreadsheet().getSheetByName(sheetName);
  if (!sheet) {
    throw new Error(`シート「${sheetName}」が見つかりません`);
  }
  return sheet;
}

/**
 * シート全体のデータを取得(オブジェクト配列で返す)
 * 1行目をヘッダーとして扱う
 * @param {string} sheetName 
 * @returns {Array<Object>} 例: [{品目ID: 'IT001', 品名: '...'}, ...]
 */
function getSheetData(sheetName) {
  const sheet = getSheet(sheetName);
  const range = sheet.getDataRange();
  const values = range.getValues();
  
  if (values.length < 2) return [];
  
  const headers = values[0];
  const rows = values.slice(1);
  
  return rows
    .filter(row => row.some(cell => cell !== ''))  // 完全な空行を除外
    .map(row => {
      const obj = {};
      headers.forEach((header, i) => {
        obj[header] = row[i];
      });
      return obj;
    });
}

/**
 * ID自動採番
 * @param {string} sheetName シート名
 * @param {string} idColumnName ID列の名前(例: '品目ID')
 * @param {string} prefix 接頭辞(例: 'IT')
 * @returns {string} 次のID(例: 'IT002')
 */
function generateNextId(sheetName, idColumnName, prefix) {
  const data = getSheetData(sheetName);
  if (data.length === 0) {
    return `${prefix}001`;
  }
  
  // 既存IDから数値部分を抽出して最大値を取得
  let maxNum = 0;
  data.forEach(row => {
    const id = String(row[idColumnName] || '');
    const match = id.match(new RegExp(`^${prefix}(\\d+)$`));
    if (match) {
      const num = parseInt(match[1], 10);
      if (num > maxNum) maxNum = num;
    }
  });
  
  // 3桁ゼロパディング、999超えたら桁数自動拡張
  const nextNum = maxNum + 1;
  const padded = String(nextNum).padStart(3, '0');
  return `${prefix}${padded}`;
}

/**
 * 日時を yyyy-MM-dd HH:mm 形式の文字列に整形
 * @param {Date} date 
 */
function formatDateTime(date) {
  if (!date) date = new Date();
  return Utilities.formatDate(date, 'Asia/Tokyo', 'yyyy-MM-dd HH:mm');
}

/**
 * 日付を yyyy-MM-dd 形式の文字列に整形
 */
function formatDate(date) {
  if (!date) date = new Date();
  return Utilities.formatDate(date, 'Asia/Tokyo', 'yyyy-MM-dd');
}

/**
 * 取引IDを生成(日付ベース + 連番)
 * 例: TR20260512-001
 */
function generateTransactionId() {
  const today = formatDate().replace(/-/g, '');
  const sheet = getSheet(SHEETS.T_INVENTORY);
  const data = getSheetData(SHEETS.T_INVENTORY);
  
  // 今日の取引IDの最大連番を取得
  let maxNum = 0;
  const prefix = `${ID_PREFIX.TRANSACTION}${today}-`;
  data.forEach(row => {
    const id = String(row['トランID'] || '');
    if (id.startsWith(prefix)) {
      const numPart = id.substring(prefix.length);
      const num = parseInt(numPart, 10);
      if (!isNaN(num) && num > maxNum) maxNum = num;
    }
  });
  
  const nextNum = String(maxNum + 1).padStart(3, '0');
  return `${prefix}${nextNum}`;
}

/**
 * 現在ログイン中のユーザーのGmailを取得
 * GAS Web App のデプロイ設定で「Execute as: User accessing」が必要
 * @returns {string} Gmail or 空文字
 */
function getCurrentUserEmail() {
  return Session.getActiveUser().getEmail() || '';
}

/**
 * Gmail から担当者情報を取得
 * @param {string} email 
 * @returns {Object|null} 担当者オブジェクト or null
 */
function findUserByEmail(email) {
  if (!email) return null;
  const users = getSheetData(SHEETS.M_USER);
  return users.find(u => u['Gmail'] === email && u['有効'] === true) || null;
}

/**
 * 認証処理(API呼び出し時の最初に必ず実行)
 * @returns {Object} 認証結果
 */
function authenticate() {
  const email = getCurrentUserEmail();
  
  if (!email) {
    return {
      authenticated: false,
      reason: 'NOT_LOGGED_IN',
      message: 'Googleアカウントでログインしてください',
    };
  }
  
  const user = findUserByEmail(email);
  
  if (!user) {
    return {
      authenticated: false,
      reason: 'NOT_REGISTERED',
      message: 'このアカウントは担当者マスターに登録されていません',
      email: email,
    };
  }
  
  // 有効=FALSE は findUserByEmail で既に除外されているため、
  // ここに到達したユーザーは全員「有効」
  
  return {
    authenticated: true,
    email: email,
    userId: user['担当者ID'],
    name: user['氏名'],
    role: user['役割'],
    isAdmin: user['役割'] === '管理者',
    isStaff: user['役割'] === '現場',
    isOffice: user['役割'] === '事務',
  };
}

/**
 * 管理者権限チェック(管理者専用APIの先頭で呼ぶ)
 * @returns {Object} 認証結果
 * @throws {Error} 管理者でない場合
 */
function requireAdmin() {
  const auth = authenticate();
  if (!auth.authenticated) {
    throw new Error(`認証エラー: ${auth.message}`);
  }
  if (!auth.isAdmin) {
    throw new Error('この操作には管理者権限が必要です');
  }
  return auth;
}

/**
 * 認証必須チェック(全API共通)
 * @returns {Object} 認証結果
 * @throws {Error} 認証失敗時
 */
function requireAuth() {
  const auth = authenticate();
  if (!auth.authenticated) {
    throw new Error(`認証エラー: ${auth.message}`);
  }
  return auth;
}

// ===== Phase 3a 追加: Gmail パラメータ認証（PWA用） =====

/**
 * Gmail パラメータで担当者を認証（PWA用簡易認証）
 * PWAから送られるgmailパラメータでM_担当者を照合する
 */
function authenticateByGmail(gmail, password) {
  if (!gmail) {
    return {
      authenticated: false,
      reason: 'NO_GMAIL',
      message: '担当者メールが未指定です',
    };
  }
  const user = findUserByEmail(gmail);
  if (!user) {
    return {
      authenticated: false,
      reason: 'NOT_REGISTERED',
      message: 'このメールアドレスは担当者マスターに登録されていません',
      email: gmail,
    };
  }
  // パスワード照合（M_担当者の「パスワード」列）
  if (!password || String(user['パスワード']) !== String(password)) {
    return {
      authenticated: false,
      reason: 'WRONG_PASSWORD',
      message: 'パスワードが正しくありません',
      email: gmail,
    };
  }
  return {
    authenticated: true,
    email: gmail,
    userId: user['担当者ID'],
    name: user['氏名'],
    role: user['役割'],
    isAdmin: user['役割'] === '管理者',
    isStaff: user['役割'] === '現場',
    isOffice: user['役割'] === '事務',
  };
}

/**
 * Gmail + パスワード必須チェック（PWA用）
 */
function requireGmailAuth(gmail, password) {
  const auth = authenticateByGmail(gmail, password);
  if (!auth.authenticated) {
    throw new Error(auth.message);
  }
  return auth;
}

/**
 * 有効な品目一覧を取得
 * カテゴリでフィルタ可能
 * @param {Object} options { category?: string }
 * @returns {Array<Object>} 品目配列
 */
function getItems(options) {
  options = options || {};
  // requireAuth(); // Phase 3a: PWAからの呼び出し対応で認証を外す
  
  const items = getSheetData(SHEETS.M_ITEM)
    .filter(item => item['有効'] === true);
  
  if (options.category) {
    return items.filter(item => item['大カテゴリ'] === options.category);
  }
  
  return items;
}

/**
 * 品目IDで1件取得
 * @param {string} itemId 
 * @returns {Object|null}
 */
function getItemById(itemId) {
  // requireAuth(); // Phase 3a: 内部関数としても使われるため認証を外す
  if (!itemId) return null;
  
  const items = getSheetData(SHEETS.M_ITEM);
  return items.find(item => item['品目ID'] === itemId) || null;
}

/**
 * バーコードで品目を検索(スキャン入力時に使用)
 * @param {string} barcode JANコード or 自社QR
 * @returns {Object|null} 品目情報 or null
 */
function findItemByBarcode(barcode) {

  if (!barcode) return null;
  
  const code = String(barcode).trim();
  if (!code) return null;
  
  const items = getSheetData(SHEETS.M_ITEM);
  
  // バーコード列で検索(完全一致、有効品のみ)
  const found = items.find(item => 
    String(item['バーコード']).trim() === code && item['有効'] === true
  );
  
  if (found) return found;
  
  // バーコード未登録の場合、品目IDでの直接検索もトライ
  // (自社QRに品目IDをそのまま埋め込んでいるケース対応)
  return items.find(item => 
    item['品目ID'] === code && item['有効'] === true
  ) || null;
}

/**
 * 大カテゴリ一覧を取得(品目選択UIのタブに使用)
 * @returns {Array<string>} カテゴリ名の配列
 */
function getCategories() {
  requireAuth();
  
  const items = getSheetData(SHEETS.M_ITEM)
    .filter(item => item['有効'] === true);
  
  const categories = new Set();
  items.forEach(item => {
    if (item['大カテゴリ']) categories.add(item['大カテゴリ']);
  });
  
  return Array.from(categories).sort();
}

/**
 * 品名で品目を検索(部分一致、大文字小文字無視)
 * @param {string} keyword 検索キーワード
 * @param {Object} options { category?: string, limit?: number }
 * @returns {Array<Object>}
 */
function searchItemsByName(keyword, options) {
  options = options || {};
  requireAuth();
  
  if (!keyword) return [];
  const kw = String(keyword).trim().toLowerCase();
  if (!kw) return [];
  
  let items = getSheetData(SHEETS.M_ITEM)
    .filter(item => item['有効'] === true);
  
  if (options.category) {
    items = items.filter(item => item['大カテゴリ'] === options.category);
  }
  
  const matched = items.filter(item => {
    const name = String(item['品名'] || '').toLowerCase();
    const subCat = String(item['中カテゴリ'] || '').toLowerCase();
    const barcode = String(item['バーコード'] || '').toLowerCase();
    const itemId = String(item['品目ID'] || '').toLowerCase();
    return name.includes(kw) || subCat.includes(kw) || 
           barcode.includes(kw) || itemId.includes(kw);
  });
  
  matched.sort((a, b) => {
    const aName = String(a['品名'] || '').toLowerCase();
    const bName = String(b['品名'] || '').toLowerCase();
    const aStartsWith = aName.startsWith(kw) ? 0 : 1;
    const bStartsWith = bName.startsWith(kw) ? 0 : 1;
    if (aStartsWith !== bStartsWith) return aStartsWith - bStartsWith;
    return aName.localeCompare(bName);
  });
  
  return matched.slice(0, options.limit || 50);
}

/**

function appendInventoryRecord(record, auth) {

/**
 * 入庫記録
 * @param {Object} input { 品目ID, 数量, 単価?, 場所ID?, 備考? }
 */
function recordStockIn(input) {
  const auth = requireAuth();
  
  // バリデーション
  if (!input.品目ID) throw new Error('品目IDが指定されていません');
  if (!input.数量 || input.数量 <= 0) {
    throw new Error('数量は正の数を指定してください(入庫)');
  }
  
  // 品目存在チェック
  const item = getItemById(input.品目ID);
  if (!item) throw new Error(`品目ID ${input.品目ID} が見つかりません`);
  if (!item['有効']) throw new Error(`品目 ${item['品名']} は無効化されています`);
  
  // 単価未指定なら品目マスターの標準単価を使う
  const unitPrice = input.単価 || item['標準単価'] || '';
  // 場所未指定なら品目マスターの主場所を使う
  const locationId = input.場所ID || item['主場所ID'] || '';
  
  return appendInventoryRecord({
    区分: '入庫',
    品目ID: input.品目ID,
    数量: input.数量,  // 入庫は正の数
    単価: unitPrice,
    場所ID: locationId,
    備考: input.備考 || '',
    ステータス: '確定',
  }, auth);
}

/**
 * 出庫(消費)記録
 * @param {Object} input { 品目ID, 数量, 場所ID?, 備考? }
 */
function recordStockOut(input) {
  const auth = requireAuth();
  
  if (!input.品目ID) throw new Error('品目IDが指定されていません');
  if (!input.数量 || input.数量 <= 0) {
    throw new Error('数量は正の数を指定してください(出庫数を入力)');
  }
  
  const item = getItemById(input.品目ID);
  if (!item) throw new Error(`品目ID ${input.品目ID} が見つかりません`);
  if (!item['有効']) throw new Error(`品目 ${item['品名']} は無効化されています`);
  
  // 在庫不足警告(エラーにはしない、現場の状況優先)
  const currentStock = getCurrentStockByItemId(input.品目ID);
  if (currentStock < input.数量) {
    Logger.log(`⚠ 在庫不足: ${item['品名']} 現在${currentStock} → 出庫${input.数量}`);
  }
  
  const locationId = input.場所ID || item['主場所ID'] || '';
  
  return appendInventoryRecord({
    区分: '出庫消費',
    品目ID: input.品目ID,
    数量: -input.数量,  // 出庫は負の数で記録
    単価: '',
    場所ID: locationId,
    備考: input.備考 || '',
    ステータス: '確定',
  }, auth);
}

  }, auth);

/**
 * 訂正承認(管理者のみ)
 * @param {string} 訂正トランID 
 */
function approveCorrection(訂正トランID) {
  const auth = requireAdmin();
  
  const lock = LockService.getScriptLock();
  if (!lock.tryLock(10000)) {
    throw new Error('他の処理が実行中です。');
  }
  
  try {
    const sheet = getSheet(SHEETS.T_INVENTORY);
    const data = sheet.getDataRange().getValues();
    const headers = data[0];
    const txnIdCol = headers.indexOf('トランID');
    const statusCol = headers.indexOf('ステータス');
    const remarksCol = headers.indexOf('備考');
    
    // 該当行を検索
    for (let i = 1; i < data.length; i++) {
      if (data[i][txnIdCol] === 訂正トランID) {
        if (data[i][statusCol] !== '申請中') {
          throw new Error(`このトランは既に処理済みです(現状: ${data[i][statusCol]})`);
        }
        // ステータスを承認済に更新
        sheet.getRange(i + 1, statusCol + 1).setValue('承認済');
        // 備考に承認情報を追記
        const oldRemarks = data[i][remarksCol] || '';
        const newRemarks = `${oldRemarks} [承認: ${auth.name} ${formatDateTime()}]`;
        sheet.getRange(i + 1, remarksCol + 1).setValue(newRemarks);
        
        return { success: true, トランID: 訂正トランID, ステータス: '承認済' };
      }
    }
    throw new Error(`トランID ${訂正トランID} が見つかりません`);
  } finally {
    lock.releaseLock();
  }
}

/**
 * 品目IDの現在在庫数を取得
 * 計算: 確定 or 承認済のレコードの数量合計、却下/申請中は除外
 * @param {string} 品目ID 
 * @returns {number}
 */
function getCurrentStockByItemId(品目ID) {
  // requireAuth(); // Phase 3a: 内部関数としても使われるため認証を外す

  const data = getSheetData(SHEETS.T_INVENTORY);
  return data
    .filter(row => row['品目ID'] === 品目ID)
    .filter(row => row['ステータス'] === '確定' || row['ステータス'] === '承認済')
    .reduce((sum, row) => sum + (Number(row['数量']) || 0), 0);
}

 */

  // requireAuth(); // Phase 3a: PWA/PC管理画面からの呼び出し対応
  
  const items = getItems();
  return items.map(item => {
    const stock = getCurrentStockByItemId(item['品目ID']);
    const threshold = Number(item['閾値']) || 0;
    return {
      品目ID: item['品目ID'],
      品名: item['品名'],
      大カテゴリ: item['大カテゴリ'],
      単位: item['単位'],
      現在在庫数: stock,
      閾値: threshold,
      アラート: threshold > 0 && stock < threshold ? '⚠ 閾値割れ' : '',
    };
  });
}

/**
 * 入出庫レコードを T_入出庫 に追記する内部関数
 * 取引後在庫を計算して同時に記録
 */
function appendInventoryRecord(record, auth) {
  const lock = LockService.getScriptLock();
  if (!lock.tryLock(10000)) {
    throw new Error('他の処理が実行中です。少し待ってから再試行してください。');
  }
  
  try {
    const sheet = getSheet(SHEETS.T_INVENTORY);
    
    const txnId = generateTransactionId();
    const now = formatDateTime();
    
    // 取引後在庫を計算(承認済みレコードのみ対象 + 今回の数量)
    let postBalance = '';
    if (record.区分 !== '訂正' || record.ステータス === '承認済') {
      // 訂正で「申請中」の場合は在庫に反映しないので空欄
      const currentStock = getCurrentStockByItemId(record.品目ID);
      postBalance = currentStock + Number(record.数量 || 0);
    }
    
    // T_入出庫の列順:
    // トランID/日時/区分/品目ID/ロットID/数量/取引後在庫/単価/場所ID/移動先場所ID/Gmail/備考/関連トランID/ステータス
    const row = [
      txnId,
      now,
      record.区分,
      record.品目ID,
      record.ロットID || '',
      record.数量,
      postBalance,
      record.単価 || '',
      record.場所ID || '',
      record.移動先場所ID || '',
      auth.email,
      record.備考 || '',
      record.関連トランID || '',
      record.ステータス || '確定',
    ];
    
    sheet.appendRow(row);
    
    return {
      success: true,
      トランID: txnId,
      日時: now,
      区分: record.区分,
      品目ID: record.品目ID,
      数量: record.数量,
      取引後在庫: postBalance,
      Gmail: auth.email,
    };
  } finally {
    lock.releaseLock();
  }
}

/**
 * 場所間移動記録
 * 1回の操作で「移動元から-N」「移動先に+N」を記録
 * 関連トランIDで2レコードを紐付け
 */
function recordStockMove(input) {
  const auth = input.gmail ? requireGmailAuth(input.gmail, input.password) : requireAuth();

  if (!input.品目ID) throw new Error('品目IDが指定されていません');
  if (!input.数量 || input.数量 <= 0) {
    throw new Error('数量は正の数を指定してください');
  }
  if (!input.場所ID) throw new Error('移動元の場所IDを指定してください');
  if (!input.移動先場所ID) throw new Error('移動先の場所IDを指定してください');
  if (input.場所ID === input.移動先場所ID) {
    throw new Error('移動元と移動先が同じです');
  }
  
  const item = getItemById(input.品目ID);
  if (!item) throw new Error(`品目ID ${input.品目ID} が見つかりません`);
  
  // 移動元から減算(数量はマイナスで記録、ただし合計在庫は変わらないので
  // 取引後在庫は計算上同じ値になる)
  // → 場所別在庫を将来追加した時のために、移動レコードは在庫合計に影響なし
  
  const lock = LockService.getScriptLock();
  if (!lock.tryLock(10000)) {
    throw new Error('他の処理が実行中です');
  }
  
  try {
    const sheet = getSheet(SHEETS.T_INVENTORY);
    const txnId1 = generateTransactionId();
    const now = formatDateTime();
    const currentStock = getCurrentStockByItemId(input.品目ID);
    
    // 移動は在庫総量を変えないので、取引後在庫は現在値のまま
    const postBalance = currentStock;
    
    // 1レコード目: 移動元から出庫
    sheet.appendRow([
      txnId1, now, '移動', input.品目ID, '',
      -input.数量, postBalance, '',
      input.場所ID, input.移動先場所ID, auth.email,
      input.備考 || `${input.場所ID}→${input.移動先場所ID} 移動`,
      '', '確定',
    ]);
    
    // 2レコード目を生成
    Utilities.sleep(100);  // タイムスタンプを変えるため
    const txnId2 = generateTransactionId();
    
    // 2レコード目: 移動先に入庫(関連トランIDで紐付け)
    sheet.appendRow([
      txnId2, formatDateTime(), '移動', input.品目ID, '',
      input.数量, postBalance, '',
      input.移動先場所ID, '', auth.email,
      input.備考 || `${input.場所ID}→${input.移動先場所ID} 移動`,
      txnId1, '確定',
    ]);
    
    return {
      success: true,
      移動元トランID: txnId1,
      移動先トランID: txnId2,
      品目ID: input.品目ID,
      数量: input.数量,
      移動元: input.場所ID,
      移動先: input.移動先場所ID,
    };
  } finally {
    lock.releaseLock();
  }
}

/**
 * 即時取消(自分が5分以内に登録したものをキャンセル)
 * 取消レコードを追加して在庫を元に戻す
 */
function revokeOwnRecord(トランID) {
  const auth = requireAuth();
  
  const original = findInventoryById(トランID);
  if (!original) throw new Error(`トランID ${トランID} が見つかりません`);
  
  // 自分のレコードか確認
  if (original['Gmail'] !== auth.email) {
    throw new Error('自分が記録したものしか取り消せません。訂正申請を使用してください。');
  }
  
  // 5分以内か確認
  const recordTime = new Date(original['日時']);
  const elapsedMinutes = (Date.now() - recordTime.getTime()) / 60000;
  if (elapsedMinutes > REVOCATION_LIMIT_MINUTES) {
    throw new Error(
      `${REVOCATION_LIMIT_MINUTES}分を超えたため即時取消できません。訂正申請を使用してください。`
    );
  }
  
  // 既に取消済み or 訂正済みでないか
  if (original['区分'] === '取消') {
    throw new Error('このレコードは既に取消です');
  }
  
  // 取消レコード追加(逆符号)
  return appendInventoryRecord({
    区分: '取消',
    品目ID: original['品目ID'],
    数量: -Number(original['数量']),
    場所ID: original['場所ID'],
    備考: `即時取消(元: ${トランID})`,
    関連トランID: トランID,
    ステータス: '確定',
  }, auth);
}

  if (!lock.tryLock(10000)) {
    throw new Error('他の処理が実行中です');
  }
  
  try {
    const sheet = getSheet(SHEETS.T_INVENTORY);
    const data = sheet.getDataRange().getValues();
    const headers = data[0];
    const txnIdCol = headers.indexOf('トランID');
    const statusCol = headers.indexOf('ステータス');
    const remarksCol = headers.indexOf('備考');
    const itemIdCol = headers.indexOf('品目ID');

        const oldRemarks = data[i][remarksCol] || '';
        const newRemarks = `${oldRemarks} [承認: ${auth.name} ${formatDateTime()}]`;
        sheet.getRange(i + 1, remarksCol + 1).setValue(newRemarks);
        
        return { success: true, トランID: 訂正トランID, ステータス: '承認済' };

/**
 * 既存レコードの取引後在庫を後付けで埋めるバッチ
 * 過去のレコードに「取引後在庫」が空欄の場合、その時点の累計を計算して埋める
 * 1回だけ実行する保守用関数
 */
function backfillPostBalance() {
  const auth = requireAdmin();
  
  const lock = LockService.getScriptLock();
  if (!lock.tryLock(30000)) {
    throw new Error('他の処理が実行中です');
  }
  
  try {
    const sheet = getSheet(SHEETS.T_INVENTORY);
    const data = sheet.getDataRange().getValues();
    const headers = data[0];
    const txnIdCol = headers.indexOf('トランID');
    const itemIdCol = headers.indexOf('品目ID');
    const qtyCol = headers.indexOf('数量');
    const postBalCol = headers.indexOf('取引後在庫');
    const statusCol = headers.indexOf('ステータス');
    const kindCol = headers.indexOf('区分');
    const dateCol = headers.indexOf('日時');
    
    // 日時順にソートするためのインデックス配列
    const indexedRows = [];
    for (let i = 1; i < data.length; i++) {
      if (!data[i][txnIdCol]) continue;  // 空行スキップ
      indexedRows.push({
        rowNum: i + 1,
        date: data[i][dateCol],
        itemId: data[i][itemIdCol],
        qty: Number(data[i][qtyCol]) || 0,
        status: data[i][statusCol],
        kind: data[i][kindCol],
        postBal: data[i][postBalCol],
      });
    }
    
    // 日時順にソート
    indexedRows.sort((a, b) => {
      if (a.date < b.date) return -1;
      if (a.date > b.date) return 1;
      return 0;
    });
    
    // 品目別に累計を計算
    const stockByItem = {};
    let updatedCount = 0;
    
    indexedRows.forEach(row => {
      // 在庫に反映するレコードか
      const affects = (row.status === '確定' || row.status === '承認済') 
                     && row.kind !== '移動';
      
      if (affects) {
        if (!stockByItem[row.itemId]) stockByItem[row.itemId] = 0;
        stockByItem[row.itemId] += row.qty;
      }
      
      // 取引後在庫が空欄なら埋める
      if ((row.postBal === '' || row.postBal === null || row.postBal === undefined) && affects) {
        sheet.getRange(row.rowNum, postBalCol + 1).setValue(stockByItem[row.itemId] || 0);
        updatedCount++;
      }
    });
    
    return {
      success: true,
      updatedCount: updatedCount,
      message: `${updatedCount}件の取引後在庫を更新しました`,
    };
  } finally {
    lock.releaseLock();
  }
}

/**
 * ロットIDを採番
 * 例: LOT001
 */
function generateLotId() {
  const sheet = getSheet(SHEETS.T_LOT || 'T_ロット');
  const data = sheet.getDataRange().getValues();
  
  let maxNum = 0;
  for (let i = 1; i < data.length; i++) {
    const id = String(data[i][0] || '');
    const match = id.match(/^LOT(\d+)$/);
    if (match) {
      const num = parseInt(match[1], 10);
      if (num > maxNum) maxNum = num;
    }
  }
  
  return `LOT${String(maxNum + 1).padStart(3, '0')}`;
}

/**

/**
 * 品目別の有効ロット一覧を取得(賞味期限の近い順)
 * @param {string} 品目ID 
 * @returns {Array<Object>}
 */
function getActiveLots(品目ID) {
  // requireAuth(); // Phase 3a: PWAからの呼び出し対応

  const lots = getSheetData('T_ロット')
    .filter(lot => lot['品目ID'] === 品目ID)
    .filter(lot => lot['ステータス'] === '有効')
    .filter(lot => Number(lot['残数量']) > 0);
  
  // 賞味期限の近い順にソート(FIFO)
  lots.sort((a, b) => {
    const dateA = new Date(a['賞味期限']);
    const dateB = new Date(b['賞味期限']);
    return dateA - dateB;
  });
  
  return lots;
}

/**
 * ロットの残数量を品目別に合計
 */
function getLotTotalByItemId(品目ID) {
  return getActiveLots(品目ID)
    .reduce((sum, lot) => sum + Number(lot['残数量']), 0);
}

/**
 * FIFO引き当て(賞味期限の近い順から自動消費)
 * 出庫数量を、賞味期限の近いロットから順に引いていく
 * @param {string} 品目ID 
 * @param {number} 必要数量 
 * @returns {Array<{ロットID, 引当数量, 残数量}>}
 */
function allocateLotsByFIFO(品目ID, 必要数量) {
  if (必要数量 <= 0) return [];
  
  const lots = getActiveLots(品目ID);
  const total = lots.reduce((sum, lot) => sum + Number(lot['残数量']), 0);
  
  if (total < 必要数量) {
    throw new Error(`ロット在庫不足: 必要${必要数量}、有効ロット合計${total}`);
  }
  
  const allocations = [];
  let remaining = 必要数量;
  
  for (const lot of lots) {
    if (remaining <= 0) break;
    
    const lotStock = Number(lot['残数量']);
    const allocate = Math.min(lotStock, remaining);
    
    allocations.push({
      ロットID: lot['ロットID'],
      ロット番号: lot['ロット番号'],
      賞味期限: lot['賞味期限'],
      引当数量: allocate,
      引当後残数量: lotStock - allocate,
    });
    
    remaining -= allocate;
  }
  
  return allocations;
}

/**
 * ロット管理品の出庫(FIFO自動引き当て + 残数量更新)
 * @param {Object} input { 品目ID, 数量, 場所ID?, 備考? }
 */
function recordStockOutWithLot(input) {
  const auth = input.gmail ? requireGmailAuth(input.gmail, input.password) : requireAuth();

  if (!input.品目ID) throw new Error('品目IDが指定されていません');
  if (!input.数量 || input.数量 <= 0) {
    throw new Error('出庫数量は正の数を指定してください');
  }
  
  const item = getItemById(input.品目ID);
  if (!item) throw new Error(`品目ID ${input.品目ID} が見つかりません`);
  
  // ロット管理品か確認
  if (!item['ロット管理']) {
    // ロット管理品でなければ通常の出庫処理
    return recordStockOut(input);
  }
  
  // FIFO引き当て
  const allocations = allocateLotsByFIFO(input.品目ID, input.数量);
  
  const lock = LockService.getScriptLock();
  if (!lock.tryLock(15000)) {
    throw new Error('他の処理が実行中です');
  }
  
  try {
    const lotSheet = getSheet('T_ロット');
    const lotData = lotSheet.getDataRange().getValues();
    const lotHeaders = lotData[0];
    const lotIdCol = lotHeaders.indexOf('ロットID');
    const remainCol = lotHeaders.indexOf('残数量');
    const statusCol = lotHeaders.indexOf('ステータス');
    
    const results = [];
    
    // ロットごとに出庫処理
    for (const alloc of allocations) {
      // T_入出庫に出庫レコード追加
      const result = appendInventoryRecord({
        区分: '出庫消費',
        品目ID: input.品目ID,
        ロットID: alloc.ロットID,
        数量: -alloc.引当数量,
        場所ID: input.場所ID || item['主場所ID'] || '',
        備考: `${input.備考 || ''} ロット:${alloc.ロット番号} 期限:${alloc.賞味期限}`.trim(),
        ステータス: '確定',
      }, auth);
      
      // T_ロットの残数量を更新
      for (let i = 1; i < lotData.length; i++) {
        if (lotData[i][lotIdCol] === alloc.ロットID) {
          lotSheet.getRange(i + 1, remainCol + 1).setValue(alloc.引当後残数量);
          // 残0になったらステータスは「有効」のまま(履歴として残す)
          // 期限切れ判定は別タイミング
          break;
        }
      }
      
      results.push(result);
    }
    
    return {
      success: true,
      品目ID: input.品目ID,
      出庫数量合計: input.数量,
      引当ロット数: allocations.length,
      引当詳細: allocations,
      records: results,
    };
  } finally {
    lock.releaseLock();
  }
}

/**
 * 期限切れアラート対象のロット一覧
 * 30日前 / 7日前 / 期限当日 の3段階で警告レベルを返す
 */
function getExpiringLots() {
  requireAuth();
  
  const today = new Date();
  today.setHours(0, 0, 0, 0);
  
  const lots = getSheetData('T_ロット')
    .filter(lot => lot['ステータス'] === '有効')
    .filter(lot => Number(lot['残数量']) > 0);
  
  const alerts = [];
  
  lots.forEach(lot => {
    const expiry = new Date(lot['賞味期限']);
    expiry.setHours(0, 0, 0, 0);
    
    const daysLeft = Math.floor((expiry - today) / (1000 * 60 * 60 * 24));
    
    let level = null;
    if (daysLeft < 0) {
      level = '🔴 期限切れ';
    } else if (daysLeft === 0) {
      level = '🔴 本日期限';
    } else if (daysLeft <= 7) {
      level = '🟡 7日以内';
    } else if (daysLeft <= 30) {
      level = '🟢 30日以内';
    }
    
    if (level) {
      const item = getItemById(lot['品目ID']);
      alerts.push({
        ロットID: lot['ロットID'],
        品目ID: lot['品目ID'],
        品名: item ? item['品名'] : '不明',
        ロット番号: lot['ロット番号'],
        賞味期限: lot['賞味期限'],
        残数量: Number(lot['残数量']),
        残日数: daysLeft,
        警告レベル: level,
      });
    }
  });
  
  // 期限の近い順にソート
  alerts.sort((a, b) => a.残日数 - b.残日数);
  
  return alerts;
}

/**
 * 期限切れロットを一括「期限切れ」ステータスに更新(管理者のみ)
 * 1日1回バッチで実行する想定
 */
function markExpiredLots() {
  const auth = requireAdmin();
  
  const today = new Date();
  today.setHours(0, 0, 0, 0);
  
  const lock = LockService.getScriptLock();
  if (!lock.tryLock(10000)) {
    throw new Error('他の処理が実行中です');
  }
  
  try {
    const sheet = getSheet('T_ロット');
    const data = sheet.getDataRange().getValues();
    const headers = data[0];
    const expiryCol = headers.indexOf('賞味期限');
    const statusCol = headers.indexOf('ステータス');
    const remainCol = headers.indexOf('残数量');
    
    let count = 0;
    for (let i = 1; i < data.length; i++) {
      const status = data[i][statusCol];
      const expiry = data[i][expiryCol];
      const remain = Number(data[i][remainCol]) || 0;
      
      if (status === '有効' && expiry && remain > 0) {
        const expiryDate = new Date(expiry);
        expiryDate.setHours(0, 0, 0, 0);
        if (expiryDate < today) {
          sheet.getRange(i + 1, statusCol + 1).setValue('期限切れ');
          count++;
        }
      }
    }
    
    return {
      success: true,
      期限切れ更新数: count,
      実行日: formatDate(),
    };
  } finally {
    lock.releaseLock();
  }
}

/**
 * 期限切れロットを廃棄処理
 * @param {string} ロットID 
 * @param {string} 廃棄理由 
 */
function discardLot(ロットID, 廃棄理由, gmail, password) {
  const auth = gmail ? requireGmailAuth(gmail, password) : requireAdmin();

  if (!ロットID) throw new Error('ロットIDを指定してください');
  if (!廃棄理由) throw new Error('廃棄理由を入力してください');
  
  const lock = LockService.getScriptLock();
  if (!lock.tryLock(10000)) {
    throw new Error('他の処理が実行中です');
  }
  
  try {
    const sheet = getSheet('T_ロット');
    const data = sheet.getDataRange().getValues();
    const headers = data[0];
    const lotIdCol = headers.indexOf('ロットID');
    const itemIdCol = headers.indexOf('品目ID');
    const remainCol = headers.indexOf('残数量');
    const statusCol = headers.indexOf('ステータス');
    
    for (let i = 1; i < data.length; i++) {
      if (data[i][lotIdCol] === ロットID) {
        const remainQty = Number(data[i][remainCol]) || 0;
        const itemId = data[i][itemIdCol];
        
        if (remainQty <= 0) {
          throw new Error('このロットは既に残数量0です');
        }
        
        // T_入出庫に「廃棄」レコード追加
        appendInventoryRecord({
          区分: '廃棄',
          品目ID: itemId,
          ロットID: ロットID,
          数量: -remainQty,
          備考: `廃棄理由: ${廃棄理由}`,
          ステータス: '確定',
        }, auth);
        
        // T_ロット を更新(残数量0、ステータス廃棄済)
        sheet.getRange(i + 1, remainCol + 1).setValue(0);
        sheet.getRange(i + 1, statusCol + 1).setValue('廃棄済');
        
        return {
          success: true,
          ロットID: ロットID,
          廃棄数量: remainQty,
          廃棄理由: 廃棄理由,
        };
      }
    }
    throw new Error(`ロットID ${ロットID} が見つかりません`);
  } finally {
    lock.releaseLock();
  }
}

/**
 * 日付セルから日本時間の Date オブジェクトを取得
 * スプシの日付セルは UTC で保存されることがあるため統一
 */
function parseSheetDate(value) {
  if (!value) return null;
  
  // すでに Date オブジェクトの場合
  if (value instanceof Date) {
    // タイムゾーン補正:日本時間として再構築
    const year = value.getFullYear();
    const month = value.getMonth();
    const date = value.getDate();
    return new Date(year, month, date, 0, 0, 0);
  }
  
  // 文字列の場合(yyyy-mm-dd 形式想定)
  if (typeof value === 'string') {
    const match = value.match(/^(\d{4})-(\d{1,2})-(\d{1,2})/);
    if (match) {
      return new Date(parseInt(match[1]), parseInt(match[2]) - 1, parseInt(match[3]), 0, 0, 0);
    }
  }
  
  return null;
}

/**
 * 全ロット一覧（品名・カテゴリを結合）
 * PC管理画面の「賞味期限」タブで使用
 */
function getAllLots() {
  const lots = getSheetData('T_ロット')
    .filter(lot => lot['ステータス'] === '有効' && Number(lot['残数量']) > 0);

  const itemCache = {};
  getSheetData(SHEETS.M_ITEM).forEach(item => {
    itemCache[item['品目ID']] = item;
  });

  return lots.map(lot => {
    const item = itemCache[lot['品目ID']] || {};
    return {
      品目ID: lot['品目ID'],
      品名: item['品名'] || '不明',
      大カテゴリ: item['大カテゴリ'] || '',
      ロットID: lot['ロットID'],
      ロット番号: lot['ロット番号'],
      残数量: Number(lot['残数量']),
      賞味期限: lot['賞味期限'] ? Utilities.formatDate(new Date(lot['賞味期限']), 'Asia/Tokyo', 'yyyy-MM-dd') : '',
      ステータス: lot['ステータス'],
    };
  }).sort((a, b) => {
    if (!a['賞味期限'] || !b['賞味期限']) return 0;
    return a['賞味期限'] < b['賞味期限'] ? -1 : 1;
  });
}

/**
 * 入出庫履歴（品名・場所名を結合）
 * PC管理画面の「入出庫履歴」タブで使用
 */
function getTransactionHistory() {
  const data = getSheetData(SHEETS.T_INVENTORY);

  const itemCache = {};
  getSheetData(SHEETS.M_ITEM).forEach(item => {
    itemCache[item['品目ID']] = item;
  });

  const locCache = {};
  getSheetData(SHEETS.M_LOCATION).forEach(loc => {
    locCache[loc['場所ID']] = loc;
  });

  return data
    .filter(row => row['ステータス'] === '確定' || row['ステータス'] === '承認済')
    .map(row => {
      const item = itemCache[row['品目ID']] || {};
      const loc = locCache[row['場所ID']] || {};
      return {
        日時: row['日時'] || '',
        区分: row['区分'] || '',
        品目ID: row['品目ID'] || '',
        品名: item['品名'] || '不明',
        大カテゴリ: item['大カテゴリ'] || '',
        数量: Number(row['数量']) || 0,
        場所名: loc['場所名'] || row['場所ID'] || '',
        Gmail: row['Gmail'] || '',
      };
    })
    .sort((a, b) => (a['日時'] > b['日時'] ? -1 : 1));
}

/**
 * GETリクエストのハンドラ(データ取得用)
 * URLの末尾に ?action=getItems のような形でパラメータを渡す
 *
 * 例:
 * https://script.google.com/.../exec?action=getCurrentStockList
 * https://script.google.com/.../exec?action=getItemById&品目ID=IT001
 */
function doGet(e) {
  return handleRequest(e);
}

/**
 * POSTリクエストのハンドラ(データ書き込み用)
 * リクエストボディに JSON を渡す
 *
 * 例:
 * POST /.../exec
 * Body: {"action":"recordStockIn","品目ID":"IT001","数量":10}
 */
function doPost(e) {
  return handleRequest(e);
}

/**
 * GET/POST 共通ハンドラ
 */
function handleRequest(e) {
  try {
    // パラメータ取得
    let params = {};

    // URLクエリパラメータ
    if (e && e.parameter) {
      params = Object.assign(params, e.parameter);
    }

    // POSTボディ(JSON)
    if (e && e.postData && e.postData.contents) {
      try {
        const body = JSON.parse(e.postData.contents);
        params = Object.assign(params, body);
      } catch (err) {
        // JSONでない場合は無視
      }
    }

    const action = params.action;
    if (!action) {
      return jsonResponse({
        success: false,
        error: 'actionパラメータが必要です',
      });
    }

    // ホワイトリスト方式: 呼び出し可能な関数を明示
    const result = dispatchAction(action, params);

    return jsonResponse({
      success: true,
      action: action,
      data: result,
    });
  } catch (err) {
    return jsonResponse({
      success: false,
      error: err.message || String(err),
      stack: err.stack || null,
    });
  }
}

/**
 * actionに応じて該当関数を呼び出す
 * セキュリティ: ホワイトリスト方式(任意の関数を呼べないようにする)
 */
function dispatchAction(action, params) {
  switch (action) {
    // === 認証 ===
    case 'authenticate':
      return authenticate();

    // === マスター取得 ===
    case 'getItems':
      return getItems({ category: params.category });
    case 'getItemById':
      return getItemById(params.品目ID);
    case 'findItemByBarcode':
      return findItemByBarcode(params.barcode);
    case 'searchItemsByName':
      return searchItemsByName(params.keyword, {
        category: params.category,
        limit: params.limit ? Number(params.limit) : undefined,
      });
    case 'getCategories':
      return getCategories();
    case 'getUsers':
      return getUsers();
    case 'getLocations':
      return getLocations();
    case 'getDiscounts':
      return getDiscounts();
    case 'getCurrentFiscalYear':
      return getCurrentFiscalYear();

    // === 在庫情報取得 ===
    case 'getCurrentStockByItemId':
      return getCurrentStockByItemId(params.品目ID);
    case 'getCurrentStockList':
      return getCurrentStockList();

    // === 入出庫(基本) ===
    case 'recordStockIn':
      return recordStockIn({
        gmail: params.gmail,
        password: params.password,
        品目ID: params.品目ID,
        数量: Number(params.数量),
        単価: params.単価 ? Number(params.単価) : undefined,
        場所ID: params.場所ID,
        備考: params.備考,
      });
    case 'recordStockOut':
      return recordStockOut({
        gmail: params.gmail,
        password: params.password,
        品目ID: params.品目ID,
        数量: Number(params.数量),
        場所ID: params.場所ID,
        備考: params.備考,
      });
    case 'recordStockMove':
      return recordStockMove({
        gmail: params.gmail,
        password: params.password,
        品目ID: params.品目ID,
        数量: Number(params.数量),
        場所ID: params.場所ID,
        移動先場所ID: params.移動先場所ID,
        備考: params.備考,
      });
    case 'recordStocktakeAdjustment':
      return recordStocktakeAdjustment({
        品目ID: params.品目ID,
        数量: Number(params.数量),
        場所ID: params.場所ID,
        備考: params.備考,
      });

    // === 取消・訂正 ===
    case 'revokeOwnRecord':
      return revokeOwnRecord(params.トランID);
    case 'requestCorrection':
      return requestCorrection({
        元トランID: params.元トランID,
        数量: Number(params.数量),
        理由: params.理由,
      });
    case 'approveCorrection':
      return approveCorrection(params.トランID);
    case 'rejectCorrection':
      return rejectCorrection(params.トランID, params.却下理由);

    // === ロット管理 ===
    case 'registerLot':
      return registerLot({
        gmail: params.gmail,
        password: params.password,
        品目ID: params.品目ID,
        ロット番号: params.ロット番号,
        仕入日: params.仕入日,
        賞味期限: params.賞味期限,
        入庫数量: Number(params.入庫数量),
        備考: params.備考,
      });
    case 'getActiveLots':
      return getActiveLots(params.品目ID);
    case 'getLotTotalByItemId':
      return getLotTotalByItemId(params.品目ID);
    case 'recordStockOutWithLot':
      return recordStockOutWithLot({
        gmail: params.gmail,
        password: params.password,
        品目ID: params.品目ID,
        数量: Number(params.数量),
        場所ID: params.場所ID,
        備考: params.備考,
      });
    case 'getExpiringLots':
      return getExpiringLots();
    case 'markExpiredLots':
      return markExpiredLots();
    case 'discardLot':
      return discardLot(params.ロットID, params.廃棄理由, params.gmail, params.password);

    // === PC管理画面用 ===
    case 'getAllLots':
      return getAllLots();
    case 'getTransactionHistory':
      return getTransactionHistory();

    // === 棚卸 ===
    case 'startStocktake':
      return startStocktake(params);
    case 'recordStocktakeCount':
      return recordStocktakeCount(params);
    case 'getStocktakeSummary':
      return getStocktakeSummary(params);
    case 'endStocktake':
      return endStocktake(params);
    case 'getStocktakeSessions':
      return getStocktakeSessions(params);
    case 'getStocktakeDetail':
      return getStocktakeDetail(params);
    case 'approveStocktake':
      return approveStocktake(params);
    case 'updateStocktakeCounts':
      return updateStocktakeCounts(params);
    case 'getMyActiveSession':
      return getMyActiveSession(params);
    case 'getUncountedItems':
      return getUncountedItems(params);
    default:
      throw new Error(`未対応のaction: ${action}`);
  }
}

/**
 * JSONレスポンスを返す
 */
function jsonResponse(obj) {
  return ContentService
    .createTextOutput(JSON.stringify(obj, null, 2))
    .setMimeType(ContentService.MimeType.JSON);
}

// ===== 1. 棚卸セッションID生成 =====

function generateStocktakeSessionId() {
  var today = formatDate().replace(/-/g, '');
  var prefix = ID_PREFIX.STOCKTAKE + today + '-';
  var sheet = getSheet('T_棚卸セッション');
  var data = sheet.getDataRange().getValues();
  var maxNum = 0;
  for (var i = 1; i < data.length; i++) {
    var id = String(data[i][0] || '');
    if (id.indexOf(prefix) === 0) {
      var num = parseInt(id.substring(prefix.length), 10);
      if (!isNaN(num) && num > maxNum) maxNum = num;
    }
  }
  return prefix + String(maxNum + 1).padStart(3, '0');
}


// ===== 2. 棚卸開始 =====

function startStocktake(params) {
  var auth = requireGmailAuth(params.gmail, params.password);

  var lock = LockService.getScriptLock();
  if (!lock.tryLock(10000)) throw new Error('他の処理が実行中です');

  try {
    var sessionId = generateStocktakeSessionId();
    var now = formatDateTime();

    var sheet = getSheet('T_棚卸セッション');
    // 列順: セッションID / 開始日時 / 担当者ID / ステータス / 終了日時 / 承認者ID / 承認日時
    sheet.appendRow([sessionId, now, auth.userId, '進行中', '', '', '']);

    // 帳簿在庫>0の全品目数
    var items = getItems();
    var totalItems = items.filter(function(item) {
      return getCurrentStockByItemId(item['品目ID']) > 0;
    }).length;

    return { sessionId: sessionId, startTime: now, totalItems: totalItems };
  } finally {
    lock.releaseLock();
  }
}


// ===== 3. 棚卸カウント記録 =====

function recordStocktakeCount(params) {
  var auth = requireGmailAuth(params.gmail, params.password);

  if (!params.sessionId) throw new Error('セッションIDが必要です');
  if (!params.品目ID) throw new Error('品目IDが必要です');
  if (params.実数 === undefined || params.実数 === '') throw new Error('実数が必要です');

  var jitsusu = Number(params.実数);
  if (isNaN(jitsusu) || jitsusu < 0) throw new Error('実数は0以上の数値を入力してください');

  // セッション存在・進行中チェック
  var sessions = getSheetData('T_棚卸セッション');
  var session = sessions.find(function(s) { return s['セッションID'] === params.sessionId; });
  if (!session) throw new Error('セッションが見つかりません');
  if (session['ステータス'] !== '進行中') throw new Error('このセッションは既に終了しています');

  // 品目情報取得
  var item = getItemById(params.品目ID);
  if (!item) throw new Error('品目ID ' + params.品目ID + ' が見つかりません');

  // 帳簿在庫取得
  var bookStock = getCurrentStockByItemId(params.品目ID);
  var diff = jitsusu - bookStock;
  var now = formatDateTime();

  var lock = LockService.getScriptLock();
  if (!lock.tryLock(10000)) throw new Error('他の処理が実行中です');

  try {
    var sheet = getSheet('T_棚卸明細');
    var data = sheet.getDataRange().getValues();
    var headers = data[0];
    var sessionIdCol = headers.indexOf('セッションID');
    var itemIdCol = headers.indexOf('品目ID');

    // 同セッション・同品目の既存行を探す（上書き用）
    var existingRow = -1;
    for (var i = 1; i < data.length; i++) {
      if (data[i][sessionIdCol] === params.sessionId && data[i][itemIdCol] === params.品目ID) {
        existingRow = i + 1; // 1-indexed
        break;
      }
    }

    // T_棚卸明細列順: セッションID / 品目ID / 帳簿在庫 / 実数 / 差異 / カウント日時 / 備考
    var row = [params.sessionId, params.品目ID, bookStock, jitsusu, diff, now, ''];

    if (existingRow > 0) {
      sheet.getRange(existingRow, 1, 1, row.length).setValues([row]);
    } else {
      sheet.appendRow(row);
    }

    return {
      品名: item['品名'],
      帳簿在庫: bookStock,
      実数: jitsusu,
      差異: diff,
    };
  } finally {
    lock.releaseLock();
  }
}


// ===== 4. 棚卸サマリー取得 =====

function getStocktakeSummary(params) {
  requireGmailAuth(params.gmail, params.password);

  if (!params.sessionId) throw new Error('セッションIDが必要です');

  var details = getSheetData('T_棚卸明細')
    .filter(function(d) { return d['セッションID'] === params.sessionId; });

  var counted = details.filter(function(d) { return d['備考'] !== '未カウント'; }).length;
  var withDifference = details.filter(function(d) {
    return d['備考'] !== '未カウント' && Number(d['差異']) !== 0;
  }).length;

  // 帳簿在庫>0の全品目数から、カウント済み品目を引く
  var items = getItems();
  var countedItemIds = {};
  details.forEach(function(d) { countedItemIds[d['品目ID']] = true; });
  var uncounted = items.filter(function(item) {
    return getCurrentStockByItemId(item['品目ID']) > 0 && !countedItemIds[item['品目ID']];
  }).length;

  // 差異あり品目の詳細リスト
  var diffDetails = details
    .filter(function(d) { return d['備考'] !== '未カウント' && Number(d['差異']) !== 0; })
    .map(function(d) {
      var item = getItemById(d['品目ID']);
      return {
        '品目ID': d['品目ID'],
        '品名': item ? item['品名'] : '',
        '帳簿在庫': d['帳簿在庫'],
        '実数': d['実数'],
        '差異': d['差異']
      };
    });

  // 全品目数
  var totalItems = items.filter(function(item) {
    return getCurrentStockByItemId(item['品目ID']) > 0;
  }).length;

  return {
    counted: counted,
    withDifference: withDifference,
    uncounted: uncounted,
    totalItems: totalItems,
    details: diffDetails
  };
}


// ===== 5. 棚卸終了 =====

function endStocktake(params) {
  var auth = requireGmailAuth(params.gmail, params.password);

  if (!params.sessionId) throw new Error('セッションIDが必要です');

  // セッション進行中チェック
  var sessionSheet = getSheet('T_棚卸セッション');
  var sessionData = sessionSheet.getDataRange().getValues();
  var sessionHeaders = sessionData[0];
  var sessionIdCol = sessionHeaders.indexOf('セッションID');
  var statusCol = sessionHeaders.indexOf('ステータス');
  var endTimeCol = sessionHeaders.indexOf('終了日時');

  var sessionRow = -1;
  for (var i = 1; i < sessionData.length; i++) {
    if (sessionData[i][sessionIdCol] === params.sessionId) {
      if (sessionData[i][statusCol] !== '進行中') throw new Error('このセッションは既に終了しています');
      sessionRow = i + 1;
      break;
    }
  }
  if (sessionRow < 0) throw new Error('セッションが見つかりません');

  var lock = LockService.getScriptLock();
  if (!lock.tryLock(10000)) throw new Error('他の処理が実行中です');

  try {
    var now = formatDateTime();

    // 未カウント品目を差異ありとして追加
    var details = getSheetData('T_棚卸明細')
      .filter(function(d) { return d['セッションID'] === params.sessionId; });
    var countedItemIds = {};
    details.forEach(function(d) { countedItemIds[d['品目ID']] = true; });

    var items = getItems();
    var detailSheet = getSheet('T_棚卸明細');
    var uncountedItems = [];

    items.forEach(function(item) {
      if (countedItemIds[item['品目ID']]) return;
      var stock = getCurrentStockByItemId(item['品目ID']);
      if (stock <= 0) return;

      // T_棚卸明細列順: セッションID / 品目ID / 帳簿在庫 / 実数 / 差異 / カウント日時 / 備考
      detailSheet.appendRow([params.sessionId, item['品目ID'], stock, '', '', now, '未カウント']);
      uncountedItems.push({ 品目ID: item['品目ID'], 品名: item['品名'], 帳簿在庫: stock });
    });

    // セッションステータスを「完了待ち」に更新
    sessionSheet.getRange(sessionRow, statusCol + 1).setValue('完了待ち');
    sessionSheet.getRange(sessionRow, endTimeCol + 1).setValue(now);

    return { success: true, uncountedItems: uncountedItems };
  } finally {
    lock.releaseLock();
  }
}


// ===== 6. 棚卸セッション一覧取得（PC管理画面用）=====

function getStocktakeSessions(params) {
  requireGmailAuth(params.gmail, params.password);

  var sessions = getSheetData('T_棚卸セッション');
  var details = getSheetData('T_棚卸明細');
  var users = getSheetData('M_担当者');

  var userCache = {};
  users.forEach(function(u) { userCache[u['担当者ID']] = u['氏名'] || u['担当者ID']; });

  return sessions.map(function(s) {
    var sid = s['セッションID'];
    var sessionDetails = details.filter(function(d) { return d['セッションID'] === sid; });
    var counted = sessionDetails.filter(function(d) { return d['備考'] !== '未カウント'; }).length;
    var withDiff = sessionDetails.filter(function(d) {
      return d['備考'] !== '未カウント' && Number(d['差異']) !== 0;
    }).length;
    var uncounted = sessionDetails.filter(function(d) { return d['備考'] === '未カウント'; }).length;

    return {
      セッションID: sid,
      開始日時: s['開始日時'] || '',
      担当者名: userCache[s['担当者ID']] || s['担当者ID'] || '',
      ステータス: s['ステータス'] || '',
      終了日時: s['終了日時'] || '',
      承認者名: s['承認者ID'] ? (userCache[s['承認者ID']] || s['承認者ID']) : '',
      承認日時: s['承認日時'] || '',
      カウント数: counted,
      差異あり数: withDiff,
      未カウント数: uncounted,
    };
  }).reverse();
}

// ===== 8. 棚卸承認（在庫調整確定）=====

function approveStocktake(params) {

  if (!params.sessionId) throw new Error('セッションIDが必要です');

  // セッション「完了待ち」チェック

  var sessionIdCol = sessionHeaders.indexOf('セッションID');

  var approverCol = sessionHeaders.indexOf('承認者ID');
  var approveTimeCol = sessionHeaders.indexOf('承認日時');

  var sessionRow = -1;
  for (var i = 1; i < sessionData.length; i++) {
    if (sessionData[i][sessionIdCol] === params.sessionId) {
      if (sessionData[i][statusCol] !== '完了待ち') throw new Error('承認できるのは「完了待ち」のセッションのみです');
      sessionRow = i + 1;
      break;
    }
  }
  if (sessionRow < 0) throw new Error('セッションが見つかりません');

  // 差異があるレコードを取得（未カウントは除く: 実数が不明なため調整不可）
  var details = getSheetData('T_棚卸明細')
    .filter(function(d) { return d['セッションID'] === params.sessionId; })
    .filter(function(d) {
      return d['備考'] !== '未カウント' && Number(d['差異']) !== 0;
    });

  var lock = LockService.getScriptLock();
  if (!lock.tryLock(10000)) throw new Error('他の処理が実行中です');

  try {
    var now = formatDateTime();
    var adjustedCount = 0;

    // 差異ありレコードごとにT_入出庫に「棚卸調整」追加
    details.forEach(function(d) {
      appendInventoryRecord({
        区分: '棚卸調整',
        品目ID: d['品目ID'],
        数量: Number(d['差異']), // 正=増、負=減（符号付き）
        備考: '棚卸セッション ' + params.sessionId,
        ステータス: '確定',
      }, auth);
      adjustedCount++;
    });

    // セッションステータスを「確定」に更新
    sessionSheet.getRange(sessionRow, statusCol + 1).setValue('確定');
    sessionSheet.getRange(sessionRow, approverCol + 1).setValue(auth.userId);
    sessionSheet.getRange(sessionRow, approveTimeCol + 1).setValue(now);

    return { adjustedCount: adjustedCount };
  } finally {
    lock.releaseLock();
  }
}


// ===== 棚卸明細の実数を一括更新（承認前修正用） =====

function updateStocktakeCounts(params) {
  var auth = requireGmailAuth(params.gmail, params.password);

  if (!params.sessionId) throw new Error('セッションIDが必要です');
  if (!params.updates) throw new Error('更新データが必要です');

  var updates;
  try {
    updates = JSON.parse(params.updates);
  } catch (e) {
    throw new Error('更新データの形式が不正です');
  }
  if (!Array.isArray(updates) || updates.length === 0) throw new Error('更新データが空です');

  // セッション存在・完了待ちチェック
  var sessions = getSheetData('T_棚卸セッション');
  var session = sessions.find(function(s) { return s['セッションID'] === params.sessionId; });
  if (!session) throw new Error('セッションが見つかりません');
  if (session['ステータス'] !== '完了待ち') throw new Error('完了待ちのセッションのみ修正できます');

  var lock = LockService.getScriptLock();
  if (!lock.tryLock(10000)) throw new Error('他の処理が実行中です');

  try {
    var sheet = getSheet('T_棚卸明細');
    var data = sheet.getDataRange().getValues();
    var headers = data[0];
    var sessionIdCol = headers.indexOf('セッションID');
    var itemIdCol = headers.indexOf('品目ID');
    var bookCol = headers.indexOf('帳簿在庫');
    var countCol = headers.indexOf('実数');
    var diffCol = headers.indexOf('差異');
    var noteCol = headers.indexOf('備考');

    var updatedCount = 0;

    for (var u = 0; u < updates.length; u++) {
      var upd = updates[u];
      var newCount = Number(upd['実数']);
      if (isNaN(newCount) || newCount < 0) continue;

      for (var i = 1; i < data.length; i++) {
        if (data[i][sessionIdCol] === params.sessionId && data[i][itemIdCol] === upd['品目ID']) {
          var rowNum = i + 1;
          var bookStock = data[i][bookCol];
          var newDiff = newCount - bookStock;
          sheet.getRange(rowNum, countCol + 1).setValue(newCount);
          sheet.getRange(rowNum, diffCol + 1).setValue(newDiff);
          // 未カウントだった場合は備考をクリア
          if (data[i][noteCol] === '未カウント') {
            sheet.getRange(rowNum, noteCol + 1).setValue('');
          }
          updatedCount++;
          break;
        }
      }
    }

    return { updatedCount: updatedCount };
  } finally {
    lock.releaseLock();
  }
}

// ===== 未カウント品目一覧を取得 =====

function getUncountedItems(params) {
  requireGmailAuth(params.gmail, params.password);

  if (!params.sessionId) throw new Error('セッションIDが必要です');

  var details = getSheetData('T_棚卸明細')
    .filter(function(d) { return d['セッションID'] === params.sessionId; });
  var countedItemIds = {};
  details.forEach(function(d) { countedItemIds[d['品目ID']] = true; });

  var items = getItems();
  var uncounted = items.filter(function(item) {
    return getCurrentStockByItemId(item['品目ID']) > 0 && !countedItemIds[item['品目ID']];
  }).map(function(item) {
    return { '品目ID': item['品目ID'], '品名': item['品名'], '大カテゴリ': item['大カテゴリ'] };
  });

  return uncounted;
}


// ===== 自分の進行中セッションを取得 =====

function getMyActiveSession(params) {
  var auth = requireGmailAuth(params.gmail, params.password);

  var sessions = getSheetData('T_棚卸セッション');
  var details = getSheetData('T_棚卸明細');
  var users = getSheetData('M_担当者');
  var userCache = {};
  users.forEach(function(u) { userCache[u['担当者ID']] = u['氏名'] || u['担当者ID']; });

  // 帳簿在庫>0の全品目数
  var items = getItems();
  var totalItems = items.filter(function(item) {
    return getCurrentStockByItemId(item['品目ID']) > 0;
  }).length;

  var mySession = null;
  var otherSessions = [];

  for (var i = 0; i < sessions.length; i++) {
    var s = sessions[i];
    if (s['ステータス'] !== '進行中') continue;

    var sid = s['セッションID'];
    var sessionDetails = details.filter(function(d) { return d['セッションID'] === sid; });
    var countedNum = sessionDetails.filter(function(d) { return d['備考'] !== '未カウント'; }).length;

    var info = {
      sessionId: sid,
      startedAt: s['開始日時'],
      担当者名: userCache[s['担当者ID']] || s['担当者ID'] || '',
      countedNum: countedNum,
      totalItems: totalItems
    };

    if (s['担当者ID'] === auth.userId) {
      mySession = info;
    } else {
      otherSessions.push(info);
    }
  }

  return {
    mySession: mySession,
    otherSessions: otherSessions
  };
}
